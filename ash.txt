static char Keypad_GetKey(void)
{
const uint16_t row_pins[4] = {
GPIO_PIN_4, GPIO_PIN_5, GPIO_PIN_6, GPIO_PIN_7
};
const uint16_t column_pins[3] = {
GPIO_PIN_0, GPIO_PIN_1, GPIO_PIN_2
};
const char keys[4][3] = {
{'1', '2', '3'},
{'4', '5', '6'},
{'7', '8', '9'},
{'', '0', '#'}
};
for (uint8_t row = 0; row < 4; row++)
{
/* Make all rows inactive /
HAL_GPIO_WritePin(GPIOA,
GPIO_PIN_4 | GPIO_PIN_5 |
GPIO_PIN_6 | GPIO_PIN_7,
GPIO_PIN_SET);
/* Activate the row being tested /
HAL_GPIO_WritePin(GPIOA, row_pins[row], GPIO_PIN_RESET);
for (uint8_t column = 0; column < 3; column++)
{
if (HAL_GPIO_ReadPin(GPIOB, column_pins[column])
== GPIO_PIN_RESET)
{
HAL_Delay(20); /* Debounce /
if (HAL_GPIO_ReadPin(GPIOB, column_pins[column])
== GPIO_PIN_RESET)
{
HAL_GPIO_WritePin(GPIOA,
GPIO_PIN_4 | GPIO_PIN_5 |
GPIO_PIN_6 | GPIO_PIN_7,
GPIO_PIN_SET);
return keys[row][column];
}
}
}
}
HAL_GPIO_WritePin(GPIOA,
GPIO_PIN_4 | GPIO_PIN_5 |
GPIO_PIN_6 | GPIO_PIN_7,
GPIO_PIN_SET);
return 0;
}

static void DWT_Delay_us(uint32_t us)
{
uint32_t start = DWT->CYCCNT;
uint32_t ticks =
us * (HAL_RCC_GetHCLKFreq() / 1000000U);
while ((DWT->CYCCNT - start) < ticks);
}
static uint8_t DHT22_WaitForState(GPIO_PinState state,
uint32_t timeout_us)
{
uint32_t start = DWT->CYCCNT;
uint32_t timeout_ticks =
timeout_us * (HAL_RCC_GetHCLKFreq() / 1000000U);
while (HAL_GPIO_ReadPin(DHT22_PORT, DHT22_PIN) != state)
{
if ((DWT->CYCCNT - start) > timeout_ticks)
return 0;
}
return 1;
}
uint8_t DHT22_Read(float temperature, float humidity)
{
uint8_t data[5] = {0};
uint32_t start;
uint32_t pulse_length;
DHT22_SetPinOutput();
HAL_GPIO_WritePin(DHT22_PORT, DHT22_PIN, GPIO_PIN_RESET);
HAL_Delay(18);
HAL_GPIO_WritePin(DHT22_PORT, DHT22_PIN, GPIO_PIN_SET);
DWT_Delay_us(30);
DHT22_SetPinInput();
if (!DHT22_WaitForState(GPIO_PIN_RESET, 100))
return 0;
if (!DHT22_WaitForState(GPIO_PIN_SET, 100))
return 0;
if (!DHT22_WaitForState(GPIO_PIN_RESET, 100))
return 0;
for (int i = 0; i < 40; i++)
{
if (!DHT22_WaitForState(GPIO_PIN_SET, 100))
return 0;
start = DWT->CYCCNT;
if (!DHT22_WaitForState(GPIO_PIN_RESET, 120))
return 0;
pulse_length =
(DWT->CYCCNT - start) /
(HAL_RCC_GetHCLKFreq() / 1000000U);
data[i / 8] <<= 1;
if (pulse_length > 50)
data[i / 8] |= 1;
}
if ((uint8_t)(data[0] + data[1] + data[2] + data[3])
!= data[4])
return 0;
humidity =
((data[0] << 8) | data[1]) / 10.0f;
uint16_t temp_raw =
((data[2] & 0x7F) << 8) | data[3];
temperature = temp_raw / 10.0f;
if (data[2] & 0x80)
temperature = -temperature;
return 1;
}

int main(void)
{

/* USER CODE END 1 /

HAL_Init();

GPIO_PIN_SET);
UART_Print("\r
System started\r
");
UART_Print("Enter password using keypad, then press #\r
");
UART_Print("* clears the password\r
");
CoreDebug-DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
DWT-CYCCNT = 0;
DWT-CTRL |= DWT_CTRL_CYCCNTENA_Msk;

/* Infinite loop /
/* USER CODE BEGIN WHILE /

{
/* USER CODE END WHILE /
/* USER CODE BEGIN 3 /
while (1)
{
char key = Keypad_GetKey();
if (key != 0)
{
if (!access_granted)
{
if (key >= '0' && key <= '9')
{
if (password_index < sizeof(entered_password) - 1)
{
entered_password[password_index++] = key;
entered_password[password_index] = '\0';
/* Hide actual password characters /
UART_Print("");
}
}
else if (key == '')
{
password_index = 0;
entered_password[0] = '\0';
UART_Print("\r
Password cleared\r
");
}
else if (key == '#')
{
UART_Print("\r
");
if (strcmp(entered_password, PASSWORD) == 0)
{
access_granted = 1;
UART_Print("Access granted\r
");
UART_Print("Sensor monitoring started\r
\r
");
/* Cause an immediate sensor reading /
last_sensor_time = HAL_GetTick() - 2000;
}
else
{
UART_Print("Wrong password\r
Try again\r
");
}
password_index = 0;
entered_password[0] = '\0';
}
}
/*
Wait for release so holding a key does not enter it
several times.
/
while (Keypad_GetKey() != 0)
{
HAL_Delay(10);
}
}
/*
Read sensors every two seconds after the correct password.
Do not use HAL_Delay(2000), because that blocks the keypad.
/
if (access_granted &&
(HAL_GetTick() - last_sensor_time >= 2000U))
{
float temperature;
float humidity;
uint32_t ldr_value;
char msg[120];
last_sensor_time = HAL_GetTick();
HAL_ADC_Start(&hadc1);
if (HAL_ADC_PollForConversion(&hadc1, 100) == HAL_OK)
{
ldr_value = HAL_ADC_GetValue(&hadc1);
}
else
{
ldr_value = 0;
}
HAL_ADC_Stop(&hadc1);
if (DHT22_Read(&temperature, &humidity))
{
int temperature_x10 = (int)(temperature * 10.0f);
int humidity_x10 = (int)(humidity * 10.0f);
snprintf(msg, sizeof(msg),
"Temperature: %d.%d C\r
"
"Humidity: %d.%d %%\r
"
"LDR ADC: %lu\r
\r
",
temperature_x10 / 10,
temperature_x10 < 0
? -(temperature_x10 % 10)
: temperature_x10 % 10,
humidity_x10 / 10,
humidity_x10 % 10,
(unsigned long)ldr_value);
UART_Print(msg);
}
else
{
snprintf(msg, sizeof(msg),
"DHT22 read error\r
"
"LDR ADC: %lu\r
\r
",
(unsigned long)ldr_value);
UART_Print(msg);
}
}
HAL_Delay(5);
}
}
/* USER CODE END 3 /
}
