#include "stm32f4xx.h"
#include <stdint.h>

/*
 * STM32F401CE keypad + virtual RFID demonstration
 *
 * Keypad:
 *   PA0-PA3 = rows (outputs)
 *   PA4-PA7 = columns (inputs with pull-ups)
 *
 * Proteus Virtual Terminal / USART1:
 *   PA9  (USART1_TX) -> Virtual Terminal RXD
 *   PA10 (USART1_RX) <- Virtual Terminal TXD
 *   9600 baud, 8 data bits, no parity, 1 stop bit
 *
 * Virtual RFID operation:
 *   Type exactly 10 characters in the Virtual Terminal and press Enter.
 *   You may also press # instead of Enter to finish the card number.
 *   Authorized demonstration card: 1234567890
 */

#define RFID_LENGTH 10U

static const char keymap[4][4] =
{
    {'7', '8', '9', '/'},
    {'4', '5', '6', '*'},
    {'1', '2', '3', '-'},
    {'C', '0', '=', '+'}
};

static const char authorizedRFID[RFID_LENGTH + 1U] = "1234567890";
static char rfidBuffer[RFID_LENGTH + 1U];
static uint8_t rfidIndex = 0U;

void delay_ms(uint32_t ms)
{
    volatile uint32_t i;

    while(ms--)
    {
        for(i = 0; i < 16000U; i++)
        {
            __NOP();
        }
    }
}

void GPIO_Init(void)
{
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;

    /* PA0-PA3: keypad row outputs. */
    GPIOA->MODER &= ~(
        (3U << (0U * 2U)) |
        (3U << (1U * 2U)) |
        (3U << (2U * 2U)) |
        (3U << (3U * 2U))
    );

    GPIOA->MODER |= (
        (1U << (0U * 2U)) |
        (1U << (1U * 2U)) |
        (1U << (2U * 2U)) |
        (1U << (3U * 2U))
    );

    /* PA4-PA7: keypad column inputs. */
    GPIOA->MODER &= ~(
        (3U << (4U * 2U)) |
        (3U << (5U * 2U)) |
        (3U << (6U * 2U)) |
        (3U << (7U * 2U))
    );

    /* Enable pull-up resistors on PA4-PA7. */
    GPIOA->PUPDR &= ~(
        (3U << (4U * 2U)) |
        (3U << (5U * 2U)) |
        (3U << (6U * 2U)) |
        (3U << (7U * 2U))
    );

    GPIOA->PUPDR |= (
        (1U << (4U * 2U)) |
        (1U << (5U * 2U)) |
        (1U << (6U * 2U)) |
        (1U << (7U * 2U))
    );

    /* Leave all keypad rows high while idle. */
    GPIOA->ODR |= (
        (1U << 0U) |
        (1U << 1U) |
        (1U << 2U) |
        (1U << 3U)
    );
}

void USART1_Init(void)
{
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

    /* PA9 and PA10: alternate-function mode. */
    GPIOA->MODER &= ~(
        (3U << (9U * 2U)) |
        (3U << (10U * 2U))
    );

    GPIOA->MODER |= (
        (2U << (9U * 2U)) |
        (2U << (10U * 2U))
    );

    /* Select AF7 (USART1) on PA9 and PA10. */
    GPIOA->AFR[1] &= ~(
        (0xFU << ((9U - 8U) * 4U)) |
        (0xFU << ((10U - 8U) * 4U))
    );

    GPIOA->AFR[1] |= (
        (7U << ((9U - 8U) * 4U)) |
        (7U << ((10U - 8U) * 4U))
    );

    /* 16 MHz peripheral clock / 9600 baud = approximately 1667. */
    USART1->BRR = 1667U;

    /* Enable transmitter, receiver and USART1. */
    USART1->CR1 = USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;
}

void USART1_SendChar(char character)
{
    while(!(USART1->SR & USART_SR_TXE))
    {
    }

    USART1->DR = (uint8_t)character;
}

void USART1_SendString(const char *text)
{
    while(*text != '\0')
    {
        USART1_SendChar(*text);
        text++;
    }
}

char USART1_GetCharNonBlocking(void)
{
    if(USART1->SR & USART_SR_RXNE)
    {
        return (char)(USART1->DR & 0xFFU);
    }

    return '\0';
}

char Keypad_GetKey(void)
{
    uint8_t row;
    uint8_t col;

    for(row = 0U; row < 4U; row++)
    {
        /* Set all rows high, then drive the current row low. */
        GPIOA->ODR |= 0x0FU;
        GPIOA->ODR &= ~(1U << row);
        delay_ms(1U);

        for(col = 0U; col < 4U; col++)
        {
            if(!(GPIOA->IDR & (1U << (col + 4U))))
            {
                char key = keymap[row][col];

                delay_ms(20U);

                while(!(GPIOA->IDR & (1U << (col + 4U))))
                {
                }

                /* Restore all rows high before returning. */
                GPIOA->ODR |= 0x0FU;
                return key;
            }
        }
    }

    GPIOA->ODR |= 0x0FU;
    return '\0';
}

uint8_t RFID_IsAuthorized(const char *received)
{
    uint8_t i;

    for(i = 0U; i < RFID_LENGTH; i++)
    {
        if(received[i] != authorizedRFID[i])
        {
            return 0U;
        }
    }

    return 1U;
}

void RFID_CheckCompletedNumber(void)
{
    if(rfidIndex == RFID_LENGTH)
    {
        rfidBuffer[RFID_LENGTH] = '\0';

        USART1_SendString("\r\nRFID card: ");
        USART1_SendString(rfidBuffer);
        USART1_SendString("\r\n");

        if(RFID_IsAuthorized(rfidBuffer))
        {
            USART1_SendString("ACCESS GRANTED\r\n");
            /* Add the door relay/servo activation here later. */
        }
        else
        {
            USART1_SendString("ACCESS DENIED\r\n");
            /* Add the alarm/buzzer activation here later. */
        }
    }
    else
    {
        USART1_SendString("\r\nINVALID RFID: enter exactly 10 characters.\r\n");
    }

    rfidIndex = 0U;
    USART1_SendString("Scan RFID: ");
}

void RFID_VirtualTerminal_Process(void)
{
    char receivedChar = USART1_GetCharNonBlocking();

    if(receivedChar == '\0')
    {
        return;
    }

    /* Ignore LF. Proteus may send CR+LF when Enter is pressed. */
    if(receivedChar == '\n')
    {
        return;
    }

    /* Enter (CR) or # completes the simulated card number. */
    if((receivedChar == '\r') || (receivedChar == '#'))
    {
        RFID_CheckCompletedNumber();
        return;
    }

    /* Support Backspace and Delete. */
    if((receivedChar == 8) || (receivedChar == 127))
    {
        if(rfidIndex > 0U)
        {
            rfidIndex--;
            USART1_SendString("\b \b");
        }

        return;
    }

    if(rfidIndex < RFID_LENGTH)
    {
        rfidBuffer[rfidIndex] = receivedChar;
        rfidIndex++;

        /* Echo the received character; keep terminal local echo off. */
        USART1_SendChar(receivedChar);
    }
    else
    {
        /* Discard extra characters until Enter/# and warn once. */
        USART1_SendChar('\a');
    }
}

int main(void)
{
    char key;

    GPIO_Init();
    USART1_Init();

    USART1_SendString("\r\n");
    USART1_SendString("STM32F401CE Smart Home Access System\r\n");
    USART1_SendString("Keypad ready.\r\n");
    USART1_SendString("Virtual RFID reader ready.\r\n");
    USART1_SendString("Authorized demo card: 1234567890\r\n");
    USART1_SendString("Type a 10-character RFID number, then Enter or #.\r\n");
    USART1_SendString("Scan RFID: ");

    while(1)
    {
        key = Keypad_GetKey();

        if(key != '\0')
        {
            USART1_SendString("\r\nKeypad key: ");
            USART1_SendChar(key);
            USART1_SendString("\r\nScan RFID: ");
        }

        RFID_VirtualTerminal_Process();
        delay_ms(10U);
    }
}
