#include "stm32f4xx.h"

void Delay_ms(volatile int ms)
{
    volatile int i, j;
    for (i = 0; i < ms; i++)
        for (j = 0; j < 3195; j++);
}

// ---------------- LCD Functions ----------------
void LCD_Pulse_EN(void)
{
    GPIOB->ODR |= (1U << 1);
    Delay_ms(1);
    GPIOB->ODR &= ~(1U << 1);
    Delay_ms(1);
}

void LCD_Send_Nibble(unsigned char nibble)
{
    GPIOB->ODR &= ~(0xF0);
    GPIOB->ODR |= ((nibble & 0x0F) << 4);
    LCD_Pulse_EN();
}

void LCD_Command(unsigned char cmd)
{
    GPIOB->ODR &= ~(1U << 0);
    LCD_Send_Nibble(cmd >> 4);
    LCD_Send_Nibble(cmd & 0x0F);
    Delay_ms(2);
}

void LCD_Data(unsigned char data)
{
    GPIOB->ODR |= (1U << 0);
    LCD_Send_Nibble(data >> 4);
    LCD_Send_Nibble(data & 0x0F);
    Delay_ms(2);
}

void LCD_Init(void)
{
    RCC->AHB1ENR |= (1U << 1);

    GPIOB->MODER &= ~(
        (3U << 0)  |
        (3U << 2)  |
        (3U << 8)  |
        (3U << 10) |
        (3U << 12) |
        (3U << 14)
    );

    GPIOB->MODER |= (
        (1U << 0)  |
        (1U << 2)  |
        (1U << 8)  |
        (1U << 10) |
        (1U << 12) |
        (1U << 14)
    );

    Delay_ms(100);

    LCD_Send_Nibble(0x03);
    Delay_ms(5);
    LCD_Send_Nibble(0x03);
    Delay_ms(1);
    LCD_Send_Nibble(0x03);
    Delay_ms(1);
    LCD_Send_Nibble(0x02);

    LCD_Command(0x28);
    LCD_Command(0x0C);
    LCD_Command(0x06);
    LCD_Command(0x01);
    Delay_ms(5);
}

void LCD_Print(char *str)
{
    while (*str)
    {
        LCD_Data((unsigned char)*str);
        str++;
    }
}

// ---------------- UART1 Functions ----------------
void UART1_Init(void)
{
    RCC->AHB1ENR |= (1U << 0);
    RCC->APB2ENR |= (1U << 4);

    GPIOA->MODER &= ~(3U << 18);
    GPIOA->MODER |=  (2U << 18);

    GPIOA->OSPEEDR |= (3U << 18);
    GPIOA->OTYPER &= ~(1U << 9);

    GPIOA->AFR[1] &= ~(0xFU << 4);
    GPIOA->AFR[1] |=  (7U << 4);

    USART1->BRR = 0x0683;
    USART1->CR1 = (1U << 3) | (1U << 13);
}

void UART1_SendChar(char c)
{
    while (!(USART1->SR & (1U << 7)));
    USART1->DR = (unsigned char)c;
}

void UART1_SendString(char *str)
{
    while (*str)
    {
        UART1_SendChar(*str);
        str++;
    }
}

// ---------------- ADC Functions ----------------
void ADC_Init(void)
{
    volatile int i;

    RCC->AHB1ENR |= (1U << 0);
    RCC->APB2ENR |= (1U << 8);

    GPIOA->MODER &= ~(3U << 0);
    GPIOA->MODER |=  (3U << 0);
    GPIOA->PUPDR &= ~(3U << 0);

    ADC->CCR &= ~(3U << 16);
    ADC->CCR |=  (1U << 16);

    ADC1->CR1 = 0;
    ADC1->CR2 = 0;
    ADC1->SQR1 = 0;
    ADC1->SQR3 = 0;

    ADC1->SMPR2 &= ~(7U << 0);
    ADC1->SMPR2 |=  (7U << 0);

    ADC1->CR2 |= (1U << 0) | (1U << 1);

    for (i = 0; i < 3000; i++);

    ADC1->CR2 |= (1U << 30);

    for (i = 0; i < 3; i++)
    {
        Delay_ms(20);
        (void)ADC1->DR;
    }
}

unsigned int ADC_Read(void)
{
    return (unsigned int)ADC1->DR;
}

// ---------------- Main Program ----------------
int main(void)
{
    unsigned int smoke_val;
    int motion_val;
    int is_smoke;
    int prev_smoke = -1;
    int prev_motion = -1;

    RCC->AHB1ENR |= (1U << 0);

    // PA1 = LED, PA2 = Buzzer Outputs
    GPIOA->MODER &= ~((3U << 2) | (3U << 4));
    GPIOA->MODER |=  ((1U << 2) | (1U << 4));
    GPIOA->OTYPER &= ~((1U << 1) | (1U << 2));
    GPIOA->ODR &= ~((1U << 1) | (1U << 2));

    // PA3 = PIR Input with Pull-Down
    GPIOA->MODER &= ~(3U << 6);
    GPIOA->PUPDR &= ~(3U << 6);
    GPIOA->PUPDR |=  (2U << 6);

    UART1_Init();
    LCD_Init();
    ADC_Init();

    Delay_ms(200);

    while (1)
    {
        smoke_val = ADC_Read();
        motion_val = (GPIOA->IDR & (1U << 3)) ? 1 : 0;
        is_smoke = (smoke_val > 3000) ? 1 : 0;

        if (is_smoke || motion_val)
        {
            GPIOA->ODR |= (1U << 1) | (1U << 2);
        }
        else
        {
            GPIOA->ODR &= ~((1U << 1) | (1U << 2));
        }

        if (is_smoke != prev_smoke || motion_val != prev_motion)
        {
            prev_smoke = is_smoke;
            prev_motion = motion_val;

            LCD_Command(0x01);
            Delay_ms(2);

            LCD_Command(0x80);
            if (is_smoke)
            {
                LCD_Print("SMOKE: DETECTED");
                UART1_SendString("ALERT: Smoke Detected!\r\n");
            }
            else
            {
                LCD_Print("SMOKE: CLEAR");
                UART1_SendString("STATUS: Air Clear.\r\n");
            }

            LCD_Command(0xC0);
            if (motion_val)
            {
                LCD_Print("MOTION: DETECTED");
                UART1_SendString("ALERT: Motion Detected!\r\n");
            }
            else
            {
                LCD_Print("MOTION: CLEAR");
                UART1_SendString("STATUS: Motion Clear.\r\n");
            }
        }

        Delay_ms(100);
    }
}
